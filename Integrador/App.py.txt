# --- Panel general ---

        "Mes": ["Abr", "May", "Jun", "Jul", "Ago", "Sep"],

# --- Participantes ---
elif menu == "Participantes":
    st.title("👩‍💼 Registro de participantes")

    nombre = st.text_input("Nombre completo")
    correo = st.text_input("Correo electrónico")
    telefono = st.text_input("Teléfono")
    rol = st.selectbox("Rol", ["Mentora", "Emprendedora", "Becada"])
    descripcion = st.text_area("Descripción profesional")

    if st.button("Guardar participante"):
        st.success(f"✅ {nombre} registrado como {rol}")

    st.subheader("Clasificación y siguiente paso")
    st.write("- **Mentora**: Validación de perfil por la coordinadora")
    st.write("- **Emprendedora**: Pago directo de membresía")
